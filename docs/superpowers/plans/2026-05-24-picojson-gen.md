# PicoJetson.Gen — Source Generator Implementation Plan

> **For agentic workers:** Use `git-commit-csharp` skill for every commit. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Implement PicoJetson.Gen Source Generator — scans `JsonSerializer.Serialize<T>` / `Deserialize<T>` calls and generates `ISerializer<T>` / `IDeserializer<T>` implementations.

**Architecture:** Single Roslyn `IIncrementalGenerator`. Scans invocation expressions matching `JsonSerializer.Serialize<T>(...)` / `JsonSerializer.Deserialize<T>(...)` patterns. For each unique `T`, generates a `file`-scoped readonly struct implementing the interface. Supports primitive types (string, int, etc.), complex types (properties → JSON key/value pairs), and `[JsonPropertyName]` attribute overrides.

**Tech Stack:** .NET Standard 2.0, Roslyn 4.x, C# latest

// PicoJetson/src/PicoJetson.Gen/JsonSerializerGenerator.cs
namespace PicoJetson.Generators;

[Generator]
public class JsonSerializerGenerator : IIncrementalGenerator
{
    private static readonly HashSet<string> BuiltInTypes = new()
    {
        "string", "int", "long", "double", "float", "bool", "byte", "short",
        "decimal", "DateTime", "DateTimeOffset", "Guid", "TimeSpan"
    };

    public void Initialize(IncrementalGeneratorInitializationContext context)
    {
        // Collect all Serialize<T> and Deserialize<T> invocations
        var invocations = context.SyntaxProvider
            .CreateSyntaxProvider(
                predicate: static (node, _) => node is InvocationExpressionSyntax
                {
                    Expression: MemberAccessExpressionSyntax
                    {
                        Name.Identifier.ValueText: "Serialize" or "SerializeToUtf8Bytes" or "Deserialize"
                    }
                },
                transform: static (ctx, _) =>
                {
                    var invocation = (InvocationExpressionSyntax)ctx.Node;
                    var ma = (MemberAccessExpressionSyntax)invocation.Expression;
                    var typeArgs = ma.Name is GenericNameSyntax gn ? gn.TypeArgumentList.Arguments : default;

                    if (typeArgs.Value.Count > 0)
                        return (ITypeSymbol?)ctx.SemanticModel.GetSymbolInfo(typeArgs.Value[0]).Symbol;

                    return null;
                })
            .Where(t => t != null)
            .Select((t, _) => t!)
            .Collect();

        context.RegisterSourceOutput(invocations, GenerateSerializer);
    }

    private static void GenerateSerializer(SourceProductionContext ctx, ImmutableArray<ITypeSymbol> types)
    {
        var generated = new HashSet<string>();

        foreach (var type in types.Distinct<ITypeSymbol>(SymbolEqualityComparer.Default))
        {
            if (!generated.Add(type.ToDisplayString())) continue;
            if (IsBuiltIn(type)) continue;

            var source = GenerateForType(type);
            ctx.AddSource($"{type.Name}.PicoJetson.g.cs", SourceText.From(source, Encoding.UTF8));
        }
    }

    private static string GenerateForType(ITypeSymbol type)
    {
        var ns = type.ContainingNamespace.ToDisplayString();
        var typeName = type.Name;
        var serializerName = $"{typeName}JsonSerializer";
        var deserializerName = $"{typeName}JsonDeserializer";
        var properties = GetSerializableProperties(type);

        var serializeBody = BuildSerializeBody(properties);
        var deserializeBody = BuildDeserializeBody(type, properties);

        return $@"// <auto-generated/>
using System;
using System.Buffers;
using System.Text;
using PicoSerDe.Abs;
using PicoJetson;

namespace {ns}
{{
    file readonly struct {serializerName} : ISerializer<{typeName}>
    {{
        public void Serialize(IBufferWriter<byte> writer, {typeName} value)
        {{
            var jw = new JsonWriter(writer);
            jw.WriteStartObject();
{serializeBody}
            jw.WriteEndObject();
        }}
    }}

    file readonly struct {deserializerName} : IDeserializer<{typeName}>
    {{
        public {typeName} Deserialize(ReadOnlySpan<byte> data)
        {{
            var reader = new JsonReader(data);
            var obj = new {typeName}();
            reader.Read();
            while (reader.Read() && reader.TokenType == TokenType.PropertyName)
            {{
                var prop = reader.GetStringRaw();
                reader.Read();
{deserializeBody}
                else
                    reader.TrySkip();
            }}
            return obj;
        }}
    }}
}}";
    }

    private static List<IPropertySymbol> GetSerializableProperties(ITypeSymbol type)
    {
        return type.GetMembers()
            .OfType<IPropertySymbol>()
            .Where(p => !p.IsStatic && !p.IsIndexer && p.GetMethod != null)
            .ToList();
    }

    private static string GetJsonPropertyName(IPropertySymbol prop)
    {
        var attr = prop.GetAttributes().FirstOrDefault(a =>
            a.AttributeClass?.Name is "JsonPropertyNameAttribute" or "JsonPropertyName");
        if (attr?.ConstructorArguments.Length > 0)
            return (string?)attr.ConstructorArguments[0].Value ?? prop.Name;
        return prop.Name;
    }

    private static string BuildSerializeBody(List<IPropertySymbol> properties)
    {
        var sb = new StringBuilder();
        foreach (var prop in properties)
        {
            var jsonName = GetJsonPropertyName(prop);
            var propType = prop.Type.ToDisplayString();

            sb.AppendLine($"            jw.WritePropertyName(\"{jsonName}\"u8);");

            if (propType == "string")
                sb.AppendLine($"            jw.WriteString(Encoding.UTF8.GetBytes(value.{prop.Name}));");
            else if (propType == "int")
                sb.AppendLine($"            jw.WriteNumber(value.{prop.Name});");
            else if (propType == "long")
                sb.AppendLine($"            jw.WriteNumber(value.{prop.Name});");
            else if (propType == "double" || propType == "float")
                sb.AppendLine($"            jw.WriteNumber(value.{prop.Name});");
            else if (propType == "bool")
                sb.AppendLine($"            jw.WriteBoolean(value.{prop.Name});");
            else
                sb.AppendLine($"            // TODO: complex type {propType}");
        }
        return sb.ToString();
    }

    private static string BuildDeserializeBody(ITypeSymbol type, List<IPropertySymbol> properties)
    {
        var sb = new StringBuilder();
        for (var i = 0; i < properties.Count; i++)
        {
            var prop = properties[i];
            var jsonName = GetJsonPropertyName(prop);
            var propType = prop.Type.ToDisplayString();
            var ifPrefix = i == 0 ? "" : "else ";

            sb.AppendLine($"                {ifPrefix}if (prop.SequenceEqual(\"{jsonName}\"u8))");

            if (propType == "string")
                sb.AppendLine($"                    obj.{prop.Name} = Encoding.UTF8.GetString(reader.GetStringRaw());");
            else if (propType == "int")
            {
                sb.AppendLine($"                {{");
                sb.AppendLine($"                    reader.TryGetInt32(out var v);");
                sb.AppendLine($"                    obj.{prop.Name} = v;");
                sb.AppendLine($"                }}");
            }
            else if (propType == "long")
            {
                sb.AppendLine($"                {{");
                sb.AppendLine($"                    reader.TryGetInt64(out var v);");
                sb.AppendLine($"                    obj.{prop.Name} = v;");
                sb.AppendLine($"                }}");
            }
            else if (propType == "double" || propType == "float")
            {
                sb.AppendLine($"                {{");
                sb.AppendLine($"                    reader.TryGetFloat64(out var v);");
                sb.AppendLine($"                    obj.{prop.Name} = ({propType})v;");
                sb.AppendLine($"                }}");
            }
            else if (propType == "bool")
            {
                sb.AppendLine($"                {{");
                sb.AppendLine($"                    reader.TryGetBool(out var v);");
                sb.AppendLine($"                    obj.{prop.Name} = v;");
                sb.AppendLine($"                }}");
            }
        }
        return sb.ToString();
    }

    private static bool IsBuiltIn(ITypeSymbol type)
    {
        return BuiltInTypes.Contains(type.Name) || type.SpecialType != SpecialType.None;
    }
}

// PicoJetson/tests/PicoJetson.Tests/JsonSerializerGeneratorTests.cs
using System.Buffers;
using System.Text;

namespace PicoJetson.Tests;

// These tests verify Source Generator output manually.
// Full Source Generator testing (CSharpSourceGeneratorVerifier) is deferred.
// This test validates the generated pattern works correctly.

public class JsonSerializerGeneratorTests
{
    public class Person
    {
        public string Name { get; set; } = "";
        public int Age { get; set; }
    }

    // Verify the SG-generated pattern (manually replicated from expected output)
    [Test]
    public async Task GeneratedSerializer_Person_RoundTrip()
    {
        var person = new Person { Name = "Alice", Age = 30 };

        // This is what the SG would generate for Person
        var serializer = new PersonSerializer();
        var deserializer = new PersonDeserializer();

        var buf = new ArrayBufferWriter<byte>(256);
        serializer.Serialize(buf, person);
        var result = deserializer.Deserialize(buf.WrittenSpan);

        await Assert.That(result.Name).IsEqualTo("Alice");
        await Assert.That(result.Age).IsEqualTo(30);
    }

    [Test]
    public async Task GeneratedSerializer_Person_ProducesValidJson()
    {
        var person = new Person { Name = "Bob", Age = 25 };
        var serializer = new PersonSerializer();
        var bytes = serializer.SerializeToBytes(person);
        var json = Encoding.UTF8.GetString(bytes);

        await Assert.That(json).IsEqualTo("{\"Name\":\"Bob\",\"Age\":25}");
    }

    // Manually generated struct (mimics SG output for validation)
    private readonly struct PersonSerializer : ISerializer<Person>
    {
        public void Serialize(IBufferWriter<byte> writer, Person value)
        {
            var jw = new JsonWriter(writer);
            jw.WriteStartObject();
            jw.WritePropertyName("Name"u8);
            jw.WriteString(Encoding.UTF8.GetBytes(value.Name));
            jw.WritePropertyName("Age"u8);
            jw.WriteNumber(value.Age);
            jw.WriteEndObject();
        }
    }

    private readonly struct PersonDeserializer : IDeserializer<Person>
    {
        public Person Deserialize(ReadOnlySpan<byte> data)
        {
            var reader = new JsonReader(data);
            var obj = new Person();
            reader.Read();
            while (reader.Read() && reader.TokenType == TokenType.PropertyName)
            {
                var prop = reader.GetStringRaw();
                reader.Read();
                if (prop.SequenceEqual("Name"u8))
                    obj.Name = Encoding.UTF8.GetString(reader.GetStringRaw());
                else if (prop.SequenceEqual("Age"u8))
                {
                    reader.TryGetInt32(out var v);
                    obj.Age = v;
                }
                else
                    reader.TrySkip();
            }
            return obj;
        }
    }
}
